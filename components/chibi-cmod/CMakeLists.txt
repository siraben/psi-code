# chibi-scheme as an ESP-IDF component.
#
# Mirrors components/lua-cmod: we don't vendor chibi's source tree
# into the repo. The build script (scripts/esp-build.sh, or
# firmware-prod's preBuild in flake.nix) clones chibi-scheme.git and
# points CHIBI_SRC_DIR at the result before invoking idf.py.
#
# CHIBI_SRC_DIR is expected to contain sexp.c / gc.c / eval.c / vm.c /
# opcodes.c / bignum.c / simplify.c plus the include/chibi headers.

if(NOT DEFINED ENV{CHIBI_SRC_DIR})
    message(FATAL_ERROR
        "CHIBI_SRC_DIR not set. Pass the path to a chibi-scheme source "
        "tree (clone of github.com/ashinn/chibi-scheme).")
endif()
set(CHIBI_SRC "$ENV{CHIBI_SRC_DIR}")

# install.h is normally generated by chibi's configure step. We don't
# run configure here because the runtime values it bakes in (module
# search path, version, etc.) aren't relevant on an embedded target —
# our embedded prompts are baked in as C string literals, not loaded
# from a module path. Generate a stub at build time so sexp.h can
# include it. This lives under the per-component build dir so it
# doesn't pollute the source tree.
set(CHIBI_GEN "${CMAKE_CURRENT_BINARY_DIR}/gen")
file(MAKE_DIRECTORY "${CHIBI_GEN}/chibi")
file(WRITE "${CHIBI_GEN}/chibi/install.h"
"/* Generated stub — chibi/install.h on embedded targets. */\n"
"#define sexp_so_extension \"\"\n"
"#define sexp_default_module_path \"\"\n"
"#define sexp_platform \"esp32\"\n"
"#define sexp_architecture \"xtensa\"\n"
"#define sexp_version \"0.11.0\"\n"
"#define sexp_release_name \"esp\"\n")

# The default chibi search path picks up install.h from the source
# tree, but we want the override above to win. Pre-create a header
# directory that combines our stub with the real chibi headers.
set(CHIBI_INC "${CMAKE_CURRENT_BINARY_DIR}/inc")
file(MAKE_DIRECTORY "${CHIBI_INC}")
# Symlink (or copy) chibi's include/ contents under inc/, then drop
# our generated install.h on top via the include-directory ordering
# (CHIBI_GEN comes first).

set(CHIBI_SOURCES
    "${CHIBI_SRC}/sexp.c"
    "${CHIBI_SRC}/gc.c"
    "${CHIBI_SRC}/eval.c"
    "${CHIBI_SRC}/vm.c"
    "${CHIBI_SRC}/opcodes.c"
    "${CHIBI_SRC}/bignum.c"
    "${CHIBI_SRC}/simplify.c"
)

idf_component_register(
    SRCS ${CHIBI_SOURCES}
    INCLUDE_DIRS "${CHIBI_GEN}" "${CHIBI_SRC}/include"
    REQUIRES lwip   # sexp.h pulls sys/socket.h unconditionally on non-PLAN9
)

# Compile flags. SEXP_USE_NO_FEATURES turns off most of the ones we
# don't need (modules, weak refs, threads, FFI, fold-case, full source
# info, simplify); we still need eval + GC + I/O. The chunk-heap
# variant keeps memory-fragmentation manageable on a single-allocator
# embedded heap.
#
# These are also exposed via target_compile_definitions(... INTERFACE
# ...) below so consumers that include chibi/eval.h see the same
# feature toggles — without that, sexp.h reads SEXP_USE_DL as truthy
# and tries to #include <dlfcn.h>.
set(CHIBI_FEATURE_DEFS
    # Keep features ON so sexp_load_standard_env can build a working
    # base environment with define / string-append / lambda. NO_FEATURES
    # is too aggressive — it leaves the globals that load_standard_env
    # touches uninitialized, and any eval that throws crashes inside
    # the exception printer.
    SEXP_USE_NO_FEATURES=0
    # No dynamic loading — newlib has no dlopen. Static libs only.
    SEXP_USE_DL=0
    # No green threads / scheduler — we run a single VM call from one
    # FreeRTOS task and discard.
    SEXP_USE_GREEN_THREADS=0
    # No FFI — too much native scaffolding.
    SEXP_USE_FFI=0
    SEXP_USE_NATIVE_X86=0
    # No bignum-via-conservative-stack tricks.
    SEXP_USE_BOEHM=0
    # Single shared heap, sized at compile time. The xtensa-esp32 chip
    # we boot on has ~120 KiB largest contiguous free block once WiFi
    # is up; 96 KiB is our budget for the GC pair.
    SEXP_USE_GLOBAL_HEAP=1
    SEXP_USE_GLOBAL_SYMBOLS=1
    SEXP_USE_FIXED_HEAP=1
    SEXP_INITIAL_HEAP_SIZE=98304
    SEXP_MAXIMUM_HEAP_SIZE=98304
    SEXP_USE_GROW_HEAP=0
    SEXP_USE_FLONUMS=1
    SEXP_USE_BIGNUMS=1
    SEXP_USE_RATIOS=0
    SEXP_USE_COMPLEX=0
    SEXP_USE_INTTYPES=1
    SEXP_USE_MUTABLE_STRINGS=0
    # Skip the runtime image-loading path — we're not loading a
    # pre-built Scheme heap.
    SEXP_USE_IMAGE_LOADING=0
    # Trim debug + safety scaffolding that costs space.
    SEXP_USE_PRINT_SOURCE_INFO=0
    SEXP_USE_FULL_SOURCE_INFO=0
    SEXP_USE_GC_FILE_DESCRIPTORS=0
    SEXP_USE_TIME_GC=0
    SEXP_USE_DEBUG_GC=0
    SEXP_USE_TYPE_PRINTERS=0
)
target_compile_definitions(${COMPONENT_LIB} PUBLIC ${CHIBI_FEATURE_DEFS})

# Some chibi macros expand to expressions that ESP-IDF treats as
# unused-value errors. Anyone including chibi/sexp.h needs the same
# warning relaxation.
target_compile_options(${COMPONENT_LIB} INTERFACE
    -Wno-error=unused-value
    -Wno-error=incompatible-pointer-types
    -Wno-error=pointer-sign
    -Wno-error=type-limits
    -Wno-error=discarded-qualifiers
)

target_compile_options(${COMPONENT_LIB} PRIVATE
    # Chibi's source has many dead branches and char-vs-int subscripts
    # that ESP-IDF's -Werror=all promotes to errors. Demote them.
    -Wno-error=unused-value
    -Wno-error=char-subscripts
    -Wno-error=unused-label
    -Wno-error=missing-field-initializers
    -Wno-error=implicit-fallthrough
    -Wno-error=incompatible-pointer-types
    -Wno-error=pointer-sign
    -Wno-error=type-limits
    -Wno-error=discarded-qualifiers
    -Wno-error=int-conversion
    -Wno-implicit-function-declaration
    -Wno-int-conversion
    -Wno-incompatible-pointer-types
    -Wno-pointer-sign
    -Wno-unused-but-set-variable
    -Wno-unused-variable
    -Wno-unused-function
    -Wno-format
    -Wno-type-limits
    -Wno-discarded-qualifiers
)
